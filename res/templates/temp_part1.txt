        const DashboardChart = ({ type, data, options }) => {
            const chartRef = useRef(null);
            const chartInstance = useRef(null);

            useEffect(() => {
                // Safely access global Chart
                const ChartConstructor = window.Chart;
                if (!ChartConstructor) {
                    console.error("Chart.js not loaded");
                    return;
                }

                if (chartRef.current) {
                    try {
                        if (chartInstance.current) chartInstance.current.destroy();

                        chartInstance.current = new ChartConstructor(chartRef.current, {
                            type: type,
                            data: data,
                            options: {
                                responsive: true,
                                maintainAspectRatio: false,
                                plugins: {
                                    legend: { display: false },
                                    title: { display: false }
                                },
                                scales: type !== 'doughnut' ? {
                                    y: { beginAtZero: true, grid: { color: 'rgba(0,0,0,0.05)' } },
                                    x: { grid: { display: false } }
                                } : {},
                                ...options
                            }
                        });
                    } catch (err) {
                        console.error("Failed to render chart:", err);
                    }
                }
                return () => { if (chartInstance.current) chartInstance.current.destroy(); };
            }, [data, options, type]);

            return <canvas ref={chartRef} style={{ width: '100%', height: '100%' }} />;
        };

                    recognition.continuous = true;
                    recognition.interimResults = true;
                    recognition.lang = 'en-US';
                    recognition.maxAlternatives = 1;

                    recognition.onstart = () => {
                        setIsMicActive(true);
                        console.log('ðŸŽ¤ Voice recognition started');

                        setIsMicActive(false);
                        setInterimTranscript('');

                        console.error('Speech recognition error:', event.error);

                        if (event.error === 'no-speech') {
                            // Auto-restart if no speech detected
                            if (recognitionRef.current && isMicActive) {
                                setTimeout(() => {
                                    try { recognition.start(); } catch (e) { console.log(e); }
                                }, 100);
                            }
                        } else if (event.error === 'network') {
                            setIsMicActive(false);
                            setInterimTranscript('');
                            alert('Network error. Please check your internet connection.');
                        } else if (event.error === 'not-allowed') {
                            setIsMicActive(false);
                            setInterimTranscript('');
                            alert('Microphone access denied. Please allow microphone access in your browser settings.');
                        } else {
                            setIsMicActive(false);
                            setInterimTranscript('');
                        }
                    };

                        let interimText = '';
                        let finalText = '';

                        for (let i = event.resultIndex; i < event.results.length; ++i) {
                            const transcript = event.results[i][0].transcript;
                            if (event.results[i].isFinal) {
                                finalText += transcript + ' ';
                            } else {
                                interimText += transcript;
                            }
                        }

                        // Update interim results for live feedback
                        if (interimText) {
                            setInterimTranscript(interimText);
                        }

                        // Add final results to answer
                        if (finalText) {
                            setCurrentAnswer(prev => {
                                const newText = prev.trim() + ' ' + finalText.trim();
                                return newText.trim();
                            });
                            setInterimTranscript('');
                        }
                    };

                    recognitionRef.current = recognition;

                    try {
                        recognition.start();
                    } catch (error) {
                        console.error('Failed to start recognition:', error);
                        alert('Failed to start voice recognition. Please try again.');
                        setIsMicActive(false);
